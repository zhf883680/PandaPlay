# 数据模型详解

## 1. ServerProfile (服务器配置)

**文件**: `src/qEmbyCore/models/profile/serverprofile.h`

```cpp
struct ServerProfile {
    enum ServerType { Emby, Jellyfin };

    // 身份
    QString id = QUuid::createUuid().toString();  // 唯一 ID
    QString name;                                  // 服务器名称
    QString url;                                   // 服务器地址
    ServerType type = Emby;                        // 服务器类型

    // SSL
    bool ignoreSslVerification = false;

    // 认证
    QString userId;        // 用户 ID
    QString userName;      // 用户名
    QString accessToken;   // 访问令牌
    QString deviceId;      // 设备 ID (UUID)

    // 权限
    bool isAdmin = false;          // 是否管理员
    bool canDownloadMedia = false; // 是否可下载

    // 图标
    QString iconBase64;   // 服务器图标 (Base64)

    bool isValid() const { return !accessToken.isEmpty(); }
};
```

```cpp
struct MediaItem {

    int criticRating = 0;             // 影评评分
    bool canDownload = false;         // 是否可下载

    // ===== 文件信息 =====
    QString container;                // 容器格式 (mkv, mp4, etc.)
    QString path;                     // 文件路径
    long long size = 0;               // 文件大小 (字节)

    // ===== 视频信息 =====
    int width = 0;                    // 宽度
    int height = 0;                   // 高度
    int partCount = 1;                // 附加部分数量
    int childCount = 0;               // 子项数量
    int recursiveItemCount = 0;       // 递归子项数量
    int parentIndexNumber = -1;       // 父级索引 (季号)
    int indexNumber = -1;             // 索引号 (集号)

    // ===== 列表信息 =====
    QStringList genres;               // 类型 (动作, 科幻, ...)
    QStringList tags;                 // 标签
    QStringList taglines;             // 宣传语
    QStringList remoteTrailers;       // 远程预告片 URL

    // ===== 嵌套结构 =====
    MediaImageInfo images;                    // 图片信息
    MediaUserDataInfo userData;               // 用户数据
    QList<MediaPersonInfo> people;            // 演职人员
    QList<MediaStudioInfo> studios;           // 制片公司
    QList<MediaExternalUrlInfo> externalUrls; // 外部链接
    QList<MediaSourceInfo> mediaSources;      // 媒体源 (视频/音频/字幕流)

    // ===== 便捷方法 =====
    bool isFavorite() const;
    QString primaryImageTag() const;
    QString backdropImageTag() const;
    QString getPrimaryImageId() const;
    static MediaItem fromJson(const QJsonObject& obj);
};
```

### 2.1 MediaImageInfo (图片信息)

```cpp
struct MediaImageInfo {
    // 直接图片标签
    QString primaryTag;            // 海报图片标签
    QString thumbTag;              // 缩略图标签
    QString backdropTag;           // 背景图标签
    QString logoTag;               // Logo 标签
    double primaryImageAspectRatio = 0.0;  // 海报宽高比
    QString primaryImageItemId;    // 海报项目 ID

    // 父级回退标签 (当自身无图时使用父级图片)
    QString parentPrimaryTag;      // 父级海报标签
    QString parentBackdropTag;     // 父级背景图标签
    QString parentThumbTag;        // 父级缩略图标签
    QString parentImageItemId;     // 父级图片项目 ID

    // 选择最佳图片 (自身优先, 无则回退到父级)
    QPair<QString, QString> bestPoster() const;    // 返回 (itemId, tag)
    QPair<QString, QString> bestThumb() const;
    QPair<QString, QString> bestBackdrop() const;
    bool isParentTag(const QString& tag) const;
};
```

**图片 URL 构造**:
```
{baseUrl}/Items/{itemId}/Images/Primary?maxWidth={maxWidth}&tag={tag}&quality={quality}
```

**图片回退逻辑**:
1. 集图片不存在 → 使用季海报
2. 季海报不存在 → 使用剧集海报
3. 缩略图不存在 → 使用海报

### 2.2 MediaUserDataInfo (用户数据)

```cpp
struct MediaUserDataInfo {
    bool isFavorite = false;              // 是否收藏
    double playedPercentage = 0.0;        // 播放百分比
    long long playbackPositionTicks = 0;  // 播放位置 (ticks)

    QString lastPlayedDate;              // 最后播放日期

```

**播放位置换算**:
- 1 tick = 100 纳秒 = 0.1 微秒
- 秒 = playbackPositionTicks / 10,000,000
- 显示时间 = ticks → hh:mm:ss

### 3.1 MediaStreamInfo (媒体流信息)

```cpp
struct MediaStreamInfo {
    // ===== 通用 =====
    QString type;           // "Video", "Audio", "Subtitle"
    QString codec;          // 编解码器: h264, h265, hevc, aac, ac3, srt, ass...
    QString codecTag;       // 编解码器标签
    QString title;          // 流标题
    QString displayTitle;   // 显示标题 (如 "English - AAC - Stereo")
    QString profile;        // 编解码器配置
    int level = 0;          // 编解码器级别
    int index = -1;         // 流索引

    // ===== 视频流 =====
    int width = 0;                  // 宽度
    int height = 0;                 // 高度
    QString aspectRatio;            // 宽高比
    bool isInterlaced = false;      // 是否隔行扫描
    float realFrameRate = 0.0f;     // 帧率
    long long bitRate = 0;          // 比特率
    int bitDepth = 0;               // 色深 (8, 10, 12)
    QString pixelFormat;            // 像素格式
    int refFrames = 0;              // 参考帧数

    // ===== 音频流 =====
    QString language;               // 语言代码 (en, zh, ja...)
    QString channelLayout;          // 声道布局 (stereo, 5.1, 7.1...)
    int channels = 0;               // 声道数
    int sampleRate = 0;             // 采样率
    bool isDefault = false;         // 是否默认

    // ===== 字幕流 =====
    QString displayLanguage;        // 显示语言名
    bool isForced = false;          // 是否强制
    bool isHearingImpaired = false; // 是否听力障碍字幕

    // ===== 外部字幕 =====
    bool isExternal = false;                // 是否外部字幕文件
    QString deliveryMethod;                 // 交付方式
    QString deliveryUrl;                    // 交付 URL
    bool isExternalUrl = false;
    bool isTextSubtitleStream = false;
    bool supportsExternalStream = false;    // 支持外挂
    QString path;                           // 字幕文件路径
    QString protocol;                       // 协议
};
```
